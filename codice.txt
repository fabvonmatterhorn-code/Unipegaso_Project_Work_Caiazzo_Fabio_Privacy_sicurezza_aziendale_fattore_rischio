      <script>
        (function() {
          const pSelect = document.getElementById('probabilita');
          const dSelect = document.getElementById('danno');
          const resDisplay = document.getElementById('risultato-calc');
          const descDisplay = document.getElementById('descrizione-rischio');

          function calcolaRischio() {
            const p = parseInt(pSelect.value);
            const d = parseInt(dSelect.value);
            const r = p * d;

            resDisplay.textContent = r;

            //colori e descrizione
            if (r <= 4) {
              resDisplay.style.color = '#2E7D32'; // Verde
              descDisplay.textContent = "Basso - Rischio Accettabile";
              descDisplay.style.color = '#2E7D32';
            } else if (r <= 8) {
              resDisplay.style.color = '#EF6C00'; // Arancio
              descDisplay.textContent = "Medio - Misure di miglioramento necessarie";
              descDisplay.style.color = '#EF6C00';
            } else {
              resDisplay.style.color = '#C62828'; // Rosso
              descDisplay.textContent = "Alto - Intervento immediato richiesto";
              descDisplay.style.color = '#C62828';
            }
          }

          pSelect.addEventListener('change', calcolaRischio);
          dSelect.addEventListener('change', calcolaRischio);
        })();
      </script>
